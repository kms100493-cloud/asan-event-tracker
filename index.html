const OWNER = 'kms100493-cloud'; const REPO = 'asan-event-tracker'; const FILE_PATH = 'data/events.md'; const HEADER_ROW = 4; // 머리글 행 const DATA_START = 5; // 데이터 시작 행 const EXCLUDE = []; // 페이지에 내보내지 않을 열 이름 const UPDATE_RE = /^(synced|updated):.*\n*/m; // 시트 어딘가의 "Claude 최근 스크리닝: yyyy-mm-dd hh:mm" 문구에서 일시를 읽음 function readStamp_(sheet){ const vals = sheet.getDataRange().getDisplayValues(); const re = /Claude 최근 스크리닝:\s*(\d{4}-\d{2}-\d{2} \d{2}:\d{2})/; for (const row of vals){ for (const v of row){ const m = String(v).match(re); if (m) return m[1]; } } return ''; } function toMarkdown_(sheet){ const values = sheet.getDataRange().getValues(); const head = values[HEADER_ROW - 1].map(v => String(v).trim()); const cols = head.map((h, i) => ({h, i})).filter(c => c.h && !EXCLUDE.includes(c.h)); const fmt = v => v instanceof Date ? Utilities.formatDate(v, 'Asia/Seoul', 'yyyy-MM-dd') : String(v == null ? '' : v).replace(/\|/g, '\\|').replace(/\r?\n/g, ' ').trim(); const lines = [ '| ' + cols.map(c => c.h).join(' | ') + ' |', '| ' + cols.map(() => '---').join(' | ') + ' |' ]; for (let r = DATA_START - 1; r < values.length; r++){ const row = values[r]; if (!String(row[0]).trim()) continue; lines.push('| ' + cols.map(c => fmt(row[c.i])).join(' | ') + ' |'); } return lines.join('\n'); } function exportToGithub(){ const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; const table = toMarkdown_(sheet); const stamp = readStamp_(sheet); const token = PropertiesService.getScriptProperties().getProperty('GITHUB_TOKEN'); const api = `https://api.github.com/repos/${OWNER}/${REPO}/contents/${FILE_PATH}`; const headers = {Authorization: 'Bearer ' + token, Accept: 'application/vnd.github+json'}; let sha = null, prevBody = null, prevStamp = null; const got = UrlFetchApp.fetch(api, {headers, muteHttpExceptions: true}); if (got.getResponseCode() === 200){ const j = JSON.parse(got.getContentText()); sha = j.sha; const text = Utilities.newBlob(Utilities.base64Decode(j.content.replace(/\n/g, ''))).getDataAsString('UTF-8'); const sm = text.match(/^updated:\s*(.*)$/m); prevStamp = sm ? sm[1].trim() : null; prevBody = text.replace(UPDATE_RE, '').trim(); } if (prevStamp !== null && prevBody === table.trim() && prevStamp === stamp){ Logger.log('변경 없음'); return; } const content = `updated: ${stamp}\n\n${table}\n`; const body = { message: `update ${stamp || '-'}`, content: Utilities.base64Encode(content, Utilities.Charset.UTF_8), branch: 'main' }; if (sha) body.sha = sha; const put = UrlFetchApp.fetch(api, { method: 'put', contentType: 'application/json', headers, payload: JSON.stringify(body), muteHttpExceptions: true }); if (put.getResponseCode() >= 300) throw new Error('GitHub 업로드 실패: ' + put.getContentText()); Logger.log('업로드 완료: ' + (stamp || '-')); } function setupTrigger(){ ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'exportToGithub') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('exportToGithub').timeBased().everyMinutes(10).create(); }
